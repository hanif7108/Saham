// --- 2. HELPER FUNCTIONS ---
get_txt_color(color bg_color) =>
    r = color.r(bg_color)
    g = color.g(bg_color)
    b = color.b(bg_color)
    luminance = (0.299 * r + 0.587 * g + 0.114 * b) / 255
    luminance > 0.5 ? color.black : color.white
